typedef struct {
    double* features;      // Array dinámico de features
    int num_features;      // Número de features
    int label;            // Etiqueta (0-4 para 5 clases)
} DataPoint;

typedef struct {
    DataPoint* data;       // Array de puntos de datos
    int num_samples;       // Número de muestras
    int num_features;      // Número de features
    int num_classes;       // Número de clases
    double* feature_means; // Media de cada feature (normalización)
    double* feature_stds;  // Desviación estándar (normalización)
} Dataset;


1.2. ARQUITECTURA DE LA RED NEURONAL
--------------------------------------------------------------------------------

typedef struct {
    int num_neurons;       // Número de neuronas en esta capa
    int num_inputs;        // Número de inputs de la capa anterior
    double** weights;      // Matriz de pesos [num_neurons][num_inputs]
    double* biases;        // Vector de biases [num_neurons]
    double* outputs;       // Salidas de activación [num_neurons]
    double* pre_activation;// Valores pre-activación (z) [num_neurons]
    double* deltas;        // Gradientes para backprop [num_neurons]
    double** weight_grads; // Gradientes de pesos [num_neurons][num_inputs]
    double* bias_grads;    // Gradientes de biases [num_neurons]
    char activation[16];   // Tipo de activación: "relu", "sigmoid", "softmax"
} Layer;

typedef struct {
    Layer** layers;        // Array de capas
    int num_layers;        // Número total de capas (incluyendo input/output)
    int input_size;        // Tamaño de entrada
    int output_size;       // Tamaño de salida (clases)
    double learning_rate;  // Tasa de aprendizaje
    double momentum;       // Momento para SGD
    double weight_decay;   // Regularización L2
    double** velocity_w;   // Velocidad para momentum en pesos
    double** velocity_b;   // Velocidad para momentum en biases
} MLP;


1.3. CONFIGURACIÓN DE ENTRENAMIENTO
--------------------------------------------------------------------------------

typedef struct {
    int epochs;            // Número de épocas de entrenamiento
    int batch_size;        // Tamaño de mini-batch
    double learning_rate;  // Learning rate inicial
    double lr_decay;       // Factor de decay por época
    double momentum;       // Momentum para SGD
    double weight_decay;   // Regularización L2
    int early_stopping;    // Epochs sin mejora antes de detener
    double min_delta;      // Mejora mínima para early stopping
    int verbose;           // Nivel de verbosidad (0, 1, 2)
    unsigned int seed;     // Semilla para reproducibilidad
} TrainConfig;


1.4. MÉTRICAS DE EVALUACIÓN
--------------------------------------------------------------------------------

typedef struct {
    double accuracy;       // Exactitud global
    double balanced_acc;   // Balanced accuracy
    double f1_weighted;    // F1-score ponderado
    double precision;      // Precisión promedio
    double recall;         // Recall promedio
    int** confusion_matrix;// Matriz de confusión [num_classes][num_classes]
    double* per_class_acc; // Accuracy por clase
    int num_classes;       // Número de clases
} Metrics;

2.4. MÓDULO: FORWARD (forward.c/h)
--------------------------------------------------------------------------------

void forward_pass(MLP* mlp, double* input);
   Input: MLP, vector de input
   Output: MLP con outputs calculados en todas las capas
   Proceso:
   1. Copiar input a primera capa
   2. Para cada capa:
      a. z = W·x + b
      b. a = activation(z)
   3. Output layer usa softmax para probabilidades

void forward_layer(Layer* layer, double* input);
   Input: Capa, input de capa anterior
   Output: Capa con outputs calculados
   Proceso:
   1. Multiplicación matriz-vector: z = W·x + b
   2. Aplicar función de activación: a = f(z)
   3. Guardar z y a para backprop


2.5. MÓDULO: BACKWARD (backward.c/h)
--------------------------------------------------------------------------------

void backward_pass(MLP* mlp, int true_label);
   Input: MLP (con forward ya ejecutado), etiqueta real
   Output: MLP con gradientes calculados
   Proceso:
   1. Calcular delta output: δ_out = pred - one_hot(label)
   2. Para cada capa desde output a input:
      a. Calcular δ_l = f'(z_l) ⊙ (W_{l+1}^T · δ_{l+1})
      b. Calcular ∇W_l = δ_l · a_{l-1}^T
      c. Calcular ∇b_l = δ_l
   3. Acumular gradientes para batch

void backward_layer(Layer* current, Layer* next, double* prev_outputs);
   Input: Capa actual, siguiente, outputs previos
   Output: Gradientes calculados en current
   Proceso:
   1. Si es última capa: δ = output - target
   2. Si no: δ = f'(z) ⊙ (W_next^T · δ_next)
   3. ∇W = δ · prev_outputs^T
   4. ∇b = δ


2.6. MÓDULO: OPTIMIZER (optimizer.c/h)
--------------------------------------------------------------------------------

void update_weights_sgd(MLP* mlp, int batch_size);
   Input: MLP con gradientes acumulados, tamaño de batch
   Output: Pesos actualizados
   Proceso:
   1. Promediar gradientes: grad /= batch_size
   2. Para cada capa:
      - W = W - lr·(grad_W + λ·W)  [con L2]
      - b = b - lr·grad_b

void update_weights_momentum(MLP* mlp, int batch_size);
   Input: MLP, batch_size
   Output: Pesos actualizados con momentum
   Proceso:
   1. v_t = β·v_{t-1} + lr·grad
   2. W_t = W_{t-1} - v_t

void zero_gradients(MLP* mlp);
   Input: MLP
   Output: Gradientes puestos a cero
   Proceso: Iterar capas y poner grads a 0

4.1. FORWARD PASS
--------------------------------------------------------------------------------
function forward_pass(mlp, input):
    current_input = input
    
    for layer in mlp.layers:
        # Linear transformation
        for i in 0 to layer.num_neurons:
            z[i] = layer.bias[i]
            for j in 0 to layer.num_inputs:
                z[i] += layer.weights[i][j] * current_input[j]
        
        # Activation
        if layer.activation == "relu":
            for i in 0 to layer.num_neurons:
                layer.outputs[i] = max(0, z[i])
        
        elif layer.activation == "sigmoid":
            for i in 0 to layer.num_neurons:
                layer.outputs[i] = 1 / (1 + exp(-z[i]))
        
        elif layer.activation == "softmax":
            max_z = max(z)
            sum_exp = 0
            for i in 0 to layer.num_neurons:
                layer.outputs[i] = exp(z[i] - max_z)
                sum_exp += layer.outputs[i]
            for i in 0 to layer.num_neurons:
                layer.outputs[i] /= sum_exp
        
        # Update for next layer
        current_input = layer.outputs


4.2. BACKWARD PASS
--------------------------------------------------------------------------------
function backward_pass(mlp, true_label):
    num_layers = mlp.num_layers
    
    # Output layer delta
    output_layer = mlp.layers[num_layers - 1]
    for i in 0 to output_layer.num_neurons:
        if i == true_label:
            output_layer.deltas[i] = output_layer.outputs[i] - 1.0
        else:
            output_layer.deltas[i] = output_layer.outputs[i]
    
    # Backpropagate through hidden layers
    for l from num_layers - 2 down to 0:
        current = mlp.layers[l]
        next = mlp.layers[l + 1]
        
        # Calculate deltas
        for i in 0 to current.num_neurons:
            sum = 0
            for j in 0 to next.num_neurons:
                sum += next.weights[j][i] * next.deltas[j]
            
            # Apply activation derivative
            if current.activation == "relu":
                derivative = (current.outputs[i] > 0) ? 1.0 : 0.0
            elif current.activation == "sigmoid":
                o = current.outputs[i]
                derivative = o * (1 - o)
            
            current.deltas[i] = sum * derivative
    
    # Calculate gradients
    for l in 0 to num_layers - 1:
        layer = mlp.layers[l]
        prev_outputs = (l == 0) ? input : mlp.layers[l-1].outputs
        
        for i in 0 to layer.num_neurons:
            layer.bias_grads[i] += layer.deltas[i]
            for j in 0 to layer.num_inputs:
                layer.weight_grads[i][j] += layer.deltas[i] * prev_outputs[j]


4.3. TRAINING LOOP
--------------------------------------------------------------------------------
function train_mlp(mlp, train_data, val_data, config):
    best_val_loss = infinity
    patience_counter = 0
    
    for epoch in 1 to config.epochs:
        shuffle_dataset(train_data, config.seed + epoch)
        
        total_loss = 0
        num_batches = ceil(train_data.num_samples / config.batch_size)
        
        for batch_idx in 0 to num_batches - 1:
            zero_gradients(mlp)
            
            start = batch_idx * config.batch_size
            end = min(start + config.batch_size, train_data.num_samples)
            batch_size = end - start
            
            # Process batch
            for i in start to end - 1:
                forward_pass(mlp, train_data.data[i].features)
                loss = -log(output[train_data.data[i].label])
                total_loss += loss
                backward_pass(mlp, train_data.data[i].label)
            
            # Update weights
            update_weights_momentum(mlp, batch_size)
        
        # Validation
        val_loss = compute_loss(mlp, val_data)
        train_loss = total_loss / train_data.num_samples
        
        print("Epoch", epoch, "Train Loss:", train_loss, "Val Loss:", val_loss)
        
        # Learning rate decay
        mlp.learning_rate *= config.lr_decay
        
        # Early stopping
        if val_loss < best_val_loss - config.min_delta:
            best_val_loss = val_loss
            patience_counter = 0
        else:
            patience_counter++
            if patience_counter >= config.early_stopping:
                print("Early stopping at epoch", epoch)
                break

5. GESTIÓN DE MEMORIA
================================================================================

Asignación dinámica requerida:
- Dataset: O(n × d) donde n=muestras, d=features
- Pesos MLP: O(Σ(L_i × L_{i+1})) para todas las capas
- Arrays temporales: O(max_layer_size)

Estrategia de liberación:
1. free_dataset(): Liberar features y arrays de datos
2. free_mlp(): Liberar capas recursivamente
3. free_metrics(): Liberar confusion matrix y arrays

Prevención de memory leaks:
- Todo malloc() tiene correspondiente free()
- Validación de malloc() != NULL
- Liberación en orden inverso a asignación
